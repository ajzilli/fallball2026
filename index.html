  <script>

    const INITIAL_ROSTER = [

      "Beckett", "Cash", "Jack B", "Max", "Dean", 

      "Arlo", "Jack N", "Ashton", "William", "Zachary"

    ];


    const POSITIONS = [

      { key: "P", label: "Pitcher", type: "IF" },

      { key: "C", label: "Catcher", type: "IF" },

      { key: "1B", label: "1st Base", type: "IF" },

      { key: "2B", label: "2nd Base", type: "IF" },

      { key: "3B", label: "3rd Base", type: "IF" },

      { key: "SS", label: "Shortstop", type: "IF" },

      { key: "LF", label: "Left Field", type: "OF" },

      { key: "LC", label: "Left Center", type: "OF" },

      { key: "RC", label: "Right Center", type: "OF" },

      { key: "RF", label: "Right Field", type: "OF" }

    ];


    const INNINGS = 6;

    let roster = [...INITIAL_ROSTER];

    // rotationState[inningIndex][posKey] = { player: "", manual: false }

    let rotationState = [];


    function init() {

      rotationState = [];

      for (let i = 0; i < INNINGS; i++) {

        const inn = {};

        POSITIONS.forEach(p => inn[p.key] = { player: "", manual: false });

        inn["BENCH"] = [];

        rotationState.push(inn);

      }

      renderRosterBadges();

      renderGrid();

      updateAudit();

    }


    function renderRosterBadges() {

      const container = document.getElementById("rosterList");

      container.innerHTML = "";

      roster.forEach((name, idx) => {

        const badge = document.createElement("span");

        badge.style.cssText = "background:#e2e8f0; padding:4px 8px; border-radius:4px; font-size:0.8rem; font-weight:600;";

        badge.innerText = `${idx + 1}. ${name}`;

        container.appendChild(badge);

      });

    }


    function renderGrid() {

      const tbody = document.getElementById("rotationBody");

      tbody.innerHTML = "";


      for (let inn = 0; inn < INNINGS; inn++) {

        const tr = document.createElement("tr");

        

        const tdInn = document.createElement("td");

        tdInn.innerHTML = `<strong>Inning ${inn + 1}</strong>`;

        tr.appendChild(tdInn);


        POSITIONS.forEach(pos => {

          const td = document.createElement("td");

          const select = document.createElement("select");

          select.className = rotationState[inn][pos.key].manual ? "manual-set" : "";


          const optDefault = document.createElement("option");

          optDefault.value = "";

          optDefault.innerText = "-- Open --";

          select.appendChild(optDefault);


          roster.forEach(player => {

            const opt = document.createElement("option");

            opt.value = player;

            opt.innerText = player;

            if (rotationState[inn][pos.key].player === player) {

              opt.selected = true;

            }

            select.appendChild(opt);

          });


          select.onchange = (e) => {

            const val = e.target.value;

            if (val === "") {

              rotationState[inn][pos.key] = { player: "", manual: false };

            } else {

              // Ensure player isn't assigned twice in same inning

              POSITIONS.forEach(p => {

                if (p.key !== pos.key && rotationState[inn][p.key].player === val) {

                  rotationState[inn][p.key] = { player: "", manual: false };

                }

              });

              rotationState[inn][pos.key] = { player: val, manual: true };

            }

            renderGrid();

            updateAudit();

          };


          td.appendChild(select);

          tr.appendChild(td);

        });


        // Calculate Bench for display

        const assignedThisInn = POSITIONS.map(p => rotationState[inn][p.key].player).filter(Boolean);

        const benchPlayers = roster.filter(p => !assignedThisInn.includes(p));

        rotationState[inn]["BENCH"] = benchPlayers;


        const tdBench = document.createElement("td");

        tdBench.style.fontWeight = "bold";

        tdBench.style.color = benchPlayers.length > 0 ? "var(--warning-text)" : "var(--text-muted)";

        tdBench.innerText = benchPlayers.length > 0 ? benchPlayers.join(", ") : "None";

        tr.appendChild(tdBench);


        tbody.appendChild(tr);

      }

    }


    function updateAudit() {

      const tbody = document.getElementById("auditBody");

      tbody.innerHTML = "";


      roster.forEach(player => {

        let totalIF = 0;

        let earlyIF = 0;

        let totalBench = 0;

        const posCounts = {};


        for (let inn = 0; inn < INNINGS; inn++) {

          let played = false;

          POSITIONS.forEach(pos => {

            if (rotationState[inn][pos.key].player === player) {

              played = true;

              posCounts[pos.key] = (posCounts[pos.key] || 0) + 1;

              if (pos.type === "IF") {

                totalIF++;

                if (inn < 4) earlyIF++;

              }

            }

          });

          if (!played) totalBench++;

        }


        const violations = [];

        if (totalIF < 3) violations.push(`Need ${3 - totalIF} more IF`);

        if (earlyIF < 2) violations.push(`Need ${2 - earlyIF} more IF (Inn 1-4)`);

        if (totalBench > 2) violations.push(`Bench limit exceeded (${totalBench}/2)`);

        

        const overusedPos = Object.entries(posCounts)

          .filter(([key, count]) => {

            const p = POSITIONS.find(pos => pos.key === key);

            return p.type === "IF" && count > 2;

          })

          .map(([key, count]) => `${key}: ${count}x`);


        if (overusedPos.length > 0) {

          violations.push(`IF duplicate: ${overusedPos.join(", ")}`);

        }


        const tr = document.createElement("tr");

        tr.innerHTML = `

          <td><strong>${player}</strong></td>

          <td>${totalIF} / 3 min</td>

          <td>${earlyIF} / 2 min</td>

          <td>${totalBench} / 2 max</td>

          <td>${Object.entries(posCounts).map(([k, c]) => `${k}:${c}`).join(" ") || "-"}</td>

          <td>

            ${violations.length === 0 

              ? `<span class="status-ok">Compliant</span>` 

              : violations.map(v => `<span class="violation-pill">${v}</span>`).join("")}

          </td>

        `;

        tbody.appendChild(tr);

      });

    }


    // Constraint-satisfaction backtracking solver for unassigned slots

    function autoFillRemaining() {

      const state = JSON.parse(JSON.stringify(rotationState));


      function isValid(innIdx, posKey, player, currentPlan) {

        const isIF = POSITIONS.find(p => p.key === posKey).type === "IF";

        

        let ifCount = 0;

        let earlyIfCount = 0;

        let samePosCount = 0;


        for (let i = 0; i < INNINGS; i++) {

          for (let p of POSITIONS) {

            if (currentPlan[i][p.key].player === player) {

              if (p.type === "IF") {

                ifCount++;

                if (i < 4) earlyIfCount++;

              }

              if (p.key === posKey) samePosCount++;

            }

          }

        }


        if (isIF && samePosCount >= 2) return false;

        return true;

      }


      function solve(innIdx, posIdx) {

        if (innIdx === INNINGS) {

          // Final check on all inning invariants

          for (let player of roster) {

            let totalIF = 0;

            let earlyIF = 0;

            let bench = 0;

            for (let i = 0; i < INNINGS; i++) {

              let pFound = false;

              for (let p of POSITIONS) {

                if (state[i][p.key].player === player) {

                  pFound = true;

                  if (p.type === "IF") {

                    totalIF++;

                    if (i < 4) earlyIF++;

                  }

                }

              }

              if (!pFound) bench++;

            }

            if (totalIF < 3 || earlyIF < 2 || bench > 2) return false;

          }

          return true;

        }


        if (posIdx === POSITIONS.length) {

          // Check bench balance: Nobody sits twice until all sat once (for 10 players, bench=0/inn)

          return solve(innIdx + 1, 0);

        }


        const pos = POSITIONS[posIdx];

        if (state[innIdx][pos.key].manual && state[innIdx][pos.key].player !== "") {

          return solve(innIdx, posIdx + 1);

        }


        // Shuffle roster to randomize assignments among valid options

        const candidates = [...roster].sort(() => Math.random() - 0.5);


        for (let cand of candidates) {

          // Check if already assigned this inning

          let assigned = false;

          for (let p of POSITIONS) {

            if (state[innIdx][p.key].player === cand) {

              assigned = true; 

              break;

            }

          }

          if (assigned) continue;


          if (isValid(innIdx, pos.key, cand, state)) {

            state[innIdx][pos.key] = { player: cand, manual: false };

            if (solve(innIdx, posIdx + 1)) return true;

            state[innIdx][pos.key] = { player: "", manual: false };

          }

        }

        return false;

      }


      if (solve(0, 0)) {

        rotationState = state;

        renderGrid();

        updateAudit();

      } else {

        alert("Could not generate a valid configuration with the current pinned manual selections. Try unlocking a few positions.");

      }

    }


    function resetToManual() {

      for (let inn = 0; inn < INNINGS; inn++) {

        POSITIONS.forEach(p => {

          if (!rotationState[inn][p.key].manual) {

            rotationState[inn][p.key] = { player: "", manual: false };

          }

        });

      }

      renderGrid();

      updateAudit();

    }


    function resetAll() {

      if (confirm("Clear all field assignments?")) {

        init();

      }

    }


    window.onload = init;
